// STEP 8: Đặt hàng bảo toàn dữ liệu bằng Transaction ACID trong EF Core
[HttpPost, Authorize]
public async Task<IActionResult> DatHang(string nguoiNhan, string diaChi, string sdt)
{
    using var transaction = await _context.Database.BeginTransactionAsync();
    try {
        var donHang = new DonDatHang { Mkh = GetCurrentUserId(), NgayDh = DateTime.Now, TenNguoiNhan = nguoiNhan, DiaChiNhan = diaChi, DienThoaiNhan = sdt };
        _context.DonDatHangs.Add(donHang); await _context.SaveChangesAsync();
        foreach (var item in cart) {
            _context.CtDatHangs.Add(new CtDatHang { Sdh = donHang.Sdh, Ms = item.Ms, SoLuong = item.SoLuong, DonGia = item.DonGia });
            var s = await _context.Saches.FindAsync(item.Ms); if (s != null) s.SoLuongBan = (s.SoLuongBan ?? 0) + item.SoLuong;
        }
        await _context.SaveChangesAsync();
        await transaction.CommitAsync(); // Cam kết Transaction thành công
        return View(""DatHangThanhCong"", donHang.Sdh);
    } catch { await transaction.RollbackAsync(); throw; }
}
